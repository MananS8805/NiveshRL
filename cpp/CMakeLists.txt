cmake_minimum_required(VERSION 3.18)
project(niveshrl_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(niveshrl_core src/core.cpp)
if(MSVC)
  target_compile_options(niveshrl_core PRIVATE /O2 /fp:precise /W3)
else()
  target_compile_options(niveshrl_core PRIVATE -O3)
endif()
install(TARGETS niveshrl_core DESTINATION .)
